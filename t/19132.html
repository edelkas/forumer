#359612 3 2007-08-26T14:44:30Z Sockmonkey
Shove this in your bots remotes:

on *:TEXT:away*:#: {
  if ($nick = usaswim) {
    if ($2 = $null) { set %away.msg No reason Given }
    else set %away.msg $2-
    set %ms On
    msg $chan usaswim is now away. Please don't disturb him. The consequences are severe. | msg $chan usaswim is away for %away.msg
  }
}
on *:Text:*usaswim*:#: {
  if (%ms = On) {
    notice $nick usaswim is away, %away.msg $+ .  If you would like to leave him a memo, type !memo <<memo>>.
  }
}
on *:text:*memo*:#: {
  set %memo $2-
  notice $nick Your memo " $+ %memo $+ " has been sent to usaswim.
  ms send usaswim %memo
}
on *:Text:back:#: {
  if ($nick = usaswim) {
    msg $chan usaswim is back from %away.msg $+ .
    set %ms Off
  }
}
