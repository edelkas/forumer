Vertex moveVec1, moveVec2, moveVec3;

moveVec2 = EdgeCollisionDetectAndResponse( pos, 1, 2, 0 );

Vertex Triangle::EdgeCollisionDetectAndResponse( Vertex & pos, int index1, int index2, int index3 )
{
Vertex edgeNormal;
Vertex projT;
Vertex projP [ 4 ];
Vertex centerOfTri;

centerOfTri.x = v[ index1 ].x + v[ index2 ].x + v[ index3 ].x;
centerOfTri.y = v[ index1 ].y + v[ index2 ].y + v[ index3 ].y;

edgeNormal.x = v[ index1 ].y - v[ index2 ].y;
edgeNormal.y = v[ index2 ].x - v[ index1 ].x;

projT = getProjection( v[ index3 ] - v[ index1 ], edgeNormal );
projP[ 0 ] = getProjection( Vertex( pos.x - 2, pos.y - 4 ) - v[ index1 ], edgeNormal );

projP[ 2 ] = getProjection( Vertex( pos.x + 2, pos.y + 4 ) - v[ index1 ], edgeNormal );
projP[ 3 ] = getProjection( Vertex( pos.x - 2, pos.y + 4 ) - v[ index1 ], edgeNormal );

return getMovementVector( projT, projP, pos - centerOfTri );
}

Vertex Triangle::getProjection ( Vertex a, Vertex b )
{
Vertex result;
float dpOverLen;

dpOverLen = ( a.x * b.x + a.y * b.y ) / ( b.x * b.x + b.y * b.y );

result.x = dpOverLen * b.x;
result.y = dpOverLen * b.y;

return result;
}
