#293528 4 2007-02-04T06:07:06Z mattk210
well, you make a movieclip of a running animation and give it the instance name of "stick"

then in the main timeline you go

function onEnterFrame(){
if (Key.isDown(Key.LEFT)){
stick._x--
}else if (Key.isDown(Key.RIGHT)){
stick._x++
}
}

what that does is to check for the left/right keys being down each frame and if they are, increment/decrement the horizontal position of the stick

#293701 6 2007-02-04T18:21:51Z jtbandes
What I like to do is have a timer instead of doing it each frame:


var frametimer = setInterval(frame, 10);

function frame() {
    if (Key.isDown(Key.LEFT)) {
        stick._x--;
    } else if (Key.isDown(Key.RIGHT)) {
        stick._x++;
    }
    
    updateAfterEvent();
}
