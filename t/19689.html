1. Factorial of a number.
Public Function Factoral(ByVal intnumber As Integer) As Integer
        Dim tempvar As Integer
        tempvar = 1
        For i As Integer = intnumber To 1 Step -1
            tempvar *= i
        Next
        Return tempvar
    End Function
