	for (var i in this.acL) {
  var p0 = this.acL[i][0]
  var p1 = this
  var p2 = this.acL[i][1]
  var dx1 = p1.x - p0.x
  var dx2 = p1.x - p2.x
  var dy1 = p1.y - p0.y
  var dy2 = p1.y - p2.y
  var dist1 = Math.sqrt(dx1*dx1+dy1*dy1)
  var minDeg = this.acL[i][2]
  var maxDeg = this.acL[i][3]

  var deg2 = new degree(p1, p2)

  minDeg = minDeg / 180 * Math.PI
  var deg = deg1.rad() - deg2.rad()

  if (deg < 0) {
  	deg += 2*Math.PI
  }
      
  var fixAngle = false
  if (deg > maxDeg) {
  	deg -= (deg-maxDeg)/2
  	fixAngle = true
  	
  } else if(deg <minDeg) {
  	
  	deg -= (deg-minDeg)/2
  	fixAngle = true
  }
    
  if (fixAngle) {
  if (deg < 0) {
  	deg += 2*Math.PI
  } else if(deg>2*Math.PI) {
  	deg -= 2*Math.PI
  }
  var finalDeg = deg2.rad() + deg
  
  p0.x = p1.x + Math.cos(finalDeg) * dist1

  p0.y = p1.y + Math.sin(finalDeg) * dist1
  var dy = p0.y - p0.oldY
  p0.oldY += dy/10
  }
  
	}
}

Here's my angular constraint code (Not Rapier's if you wanted to know, his is way too stiff to be mine). Needs a couple tweeks to fix a sudden change from 360 to 0 so that the minimum angle is not constrained, but the maximum is. Works perfectly other than that.

Particle.prototype.addAngStick = function(Leg1, Vertex, Leg2, MinAng, MaxAng) {
	this.apList = new Array();
	this.angList = new Array();
	this.apList.push(Leg1);
	this.apList.push(Vertex);
	this.apList.push(Leg2);
	if (_root.hrgt) {
 this.angList.push(360-MaxAng);
 this.angList.push(360-MinAng);
	} else {
 this.angList.push(MinAng);
 this.angList.push(MaxAng);
	}
};
Particle.prototype.arelax = function() {
	if (this.apList != undefined) {
 var apL = this.apList;
 var angL = this.angList;
 var l0 = apL[0];
 var l1 = apL[1];
 var l2 = apL[2];
 var MINA = angL[0];
 var MAXA = angL[1];
 var dx1 = l0.x-l1.x;
 var dy1 = l0.y-l1.y;
 var dx2 = l2.x-l1.x;
 var dy2 = l2.y-l1.y;
 var octx = (l0.x+l1.x+l2.x)/3;
 var octy = (l0.y+l1.y+l2.y)/3;
 var dist1 = Math.sqrt(dx1*dx1+dy1*dy1);
 var dist2 = Math.sqrt(dx2*dx2+dy2*dy2);
 var ang1 = Math.atan2(dy1, dx1);
 var ang2 = Math.atan2(dy2, dx2);
 var ANG = (ang1-ang2)*(180/Math.PI);
 if (ANG<0) {
 	ANG += 360;
 }
 if (ANG<MINA) {
 	var dAng = (MINA-ANG)/180/Math.PI/2;
 	l0.x = (dist1*Math.cos(ang1+dAng)+l1.x);
 	l0.y = (dist1*Math.sin(ang1+dAng)+l1.y);
 	l2.x = (dist2*Math.cos(ang2-dAng)+l1.x);
 	l2.y = (dist2*Math.sin(ang2-dAng)+l1.y);
 } else if (ANG>MAXA) {
 	var dAng = (MAXA-ANG)/180/Math.PI/2;
 	l0.x = (dist1*Math.cos(ang1+dAng)+l1.x);
 	l0.y = (dist1*Math.sin(ang1+dAng)+l1.y);
 	l2.x = (dist2*Math.cos(ang2-dAng)+l1.x);
 	l2.y = (dist2*Math.sin(ang2-dAng)+l1.y);
 }
 var nctx = (l0.x+l1.x+l2.x)/3;
 var ncty = (l0.y+l1.y+l2.y)/3;
 var ctx = nctx-octx;
 var cty = ncty-octy;
 l0.x -= ctx;
 l0.y -= cty;
 l1.x -= ctx;
 l1.y -= cty;
 l2.x -= ctx;
 l2.y -= cty;
	}
};
