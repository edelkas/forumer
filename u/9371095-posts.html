What you've coded work only if a square A put its dx and dy (and rotates in accordance with
dx and dy with atan2) acording to the deltax and deltay of a square B.
boxA.xw = boxA._width / 2;
boxA.yw = boxA._height / 2;
boxA.dx = 1;
boxA.dy = 0;
boxB.xw = boxB._width / 2;
boxB.yw = boxB._height / 2;
boxB.dx = 1;
boxB.dy = 0;
//->boxB._rotation = Math.atan2 (boxB.dy, boxB.dx) * 180 / Math.PI;
function ProjectBox (box, ax, ay) {
	ix = box.xw * box.dx;
	iy = box.xw * box.dy;
	jx = box.yw * -box.dy;
	jy = box.yw * box.dx;
	dpi = ix * ax + iy * ay;
	dpj = jx * ax + jy * ay;
	return Math.abs (dpi) + Math.abs (dpj);
}
this.onEnterFrame = function () {
	txtMessage.text = "";
	boxA.px = boxA._x;
	boxA.py = boxA._y;
	boxB.px = boxB._x;
	boxB.py = boxB._y;
	deltax = boxA.px - boxB.px;
	deltay = boxA.py - boxB.py;
	len = Math.sqrt (deltax * deltax + deltay * deltay);
	boxB.dx = deltax / len;
	boxB.dy = deltay / len;
	//->boxB._rotation = Math.atan2 (boxB.dy, boxB.dx) * 180 / Math.PI;
	axis_x = boxB.dx;
	axis_y = boxB.dy;
	asize = ProjectBox (boxA, axis_x, axis_y);
	bsize = boxB.xw;
	dsize = Math.abs (deltax * axis_x + deltay * axis_y);
	penAx = (asize + bsize) - dsize;
	if (penAx > 0) {
  // boxes overlap along axis;  check next axis
  txtMessage.text = "1st x check passed... checking 1st y";
  axis_x = boxB.dy;
  axis_y = -boxB.dx;
  asize = ProjectBox (boxA, axis_x, axis_y);
  bsize = boxB.yw;
  dsize = Math.abs (deltax * axis_x + deltay * axis_y);
  penAy = (asize + bsize) - dsize;
  if (penAy > 0) {
  	// boxes overlap along axis;  check next axis
  	txtMessage.text += '\n1st y check passed...checking 2nd x';
  	axis_x = boxA.dx;
  	axis_y = boxA.dy;
  	asize = boxA.xw;
  	bsize = ProjectBox (boxB, axis_x, axis_y);
  	dsize = Math.abs (deltax * axis_x + deltay * axis_y);
  	penBx = (asize + bsize) - dsize;
  	if (penBx > 0) {
    // boxes overlap along axis;  check next axis
