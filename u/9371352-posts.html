Yes and a warning: GAME MAKER IS BLOODY HARD TO USE.
Its easy... 
if keyboard_check_direct(vk_left)=false
{if shiprotspeed>0 
{shiprotspeed=(shiprotspeed/1.1)
if shiprotspeed<2 then shiprotspeed=0}
}
if keyboard_check_direct(vk_right)=false
{if shiprotspeed<0 {shiprotspeed=(shiprotspeed/1.1)
if shiprotspeed>-2 then shiprotspeed=0}
}

if shiprotspeed>44 then shiprotspeed=44
if shiprotspeed<-44 then shiprotspeed=-44
shipdir=shipdir+shiprotspeed

if speed>8 then speed=8
image_index=shipdir/5

if reload<100 then reload+=1
if !keyboard_check(ord('k'))
{
if abf<1 then recharge=true
if recharge=true
{abf+=1}
}
if abf=1000 recharge=false
See? perfectly simple.
