function itemConstruct(obj)
{
obj.setting.suspend=true;
obj.setting.pickUp=true;

// model

obj.model.on=true;
obj.model.name='Atom';
obj.model.lit=DIM3_MODEL_LIT_VERTEX;
obj.model.shadow.on=true;

// settings

obj.size.x=600;
obj.size.z=600;
obj.size.y=600;
}

//
// item pickup callback
//

function itemPickup(obj)
{
// pickup the item

sound.play("Got Ammo",obj.position.x,obj.position.z,obj.position.y,1);
obj.pickup.addHealth(obj.pickup.objectId,10000);
map.action.setMap('Fight 2','Start','Player');

// turn off pickups and fade out

obj.setting.contact=false;
obj.event.startTimer(1,0);
}

//
// item fades
//

function itemFadeOut(obj,tick)
{
obj.model.alpha-=0.1;
obj.model.resize-=0.08;
if (obj.model.alpha>0) return;

// all the way faded out

obj.model.alpha=0;
obj.setting.hidden=true;

// wait for fade in

obj.event.startTimer(50000000000000000000,1000000000000);
}

function itemFadeIn(obj,tick)
{
obj.model.alpha+=0.1;
obj.model.resize+=0.08;
if (obj.model.alpha<1) return;

// all the way faded in

obj.model.alpha=1;
obj.model.resize=1;

// ready for pickup

obj.setting.contact=true;
obj.event.clearTimer();
}

function itemRespawn(obj,tick)
{
// play teleport sound

sound.play("Got Ammo",obj.position.x,obj.position.z,obj.position.y,1);

// start fade in

obj.setting.hidden=false;

obj.event.startTimer(1,2);
}

function itemTimer(obj,id)
{
switch (id) {

case 0:
itemFadeOut(obj);
return;

case 1:
itemRespawn(obj);
return;

case 2:
itemFadeIn(obj);
return;

}
}

//
// item events
//

function event(obj,mainEvent,subEvent,id,tick)
{
switch (mainEvent) {

case DIM3_EVENT_CONSTRUCT:
itemConstruct(obj);
return;

case DIM3_EVENT_PICKUP:
itemPickup(obj);
return;

case DIM3_EVENT_TIMER:
itemTimer(obj,id);
return;


}
}
