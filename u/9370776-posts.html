@Sephorith:
// Force of gravity.
var g = 3;

// Force of movement.
var m = 5;

// Force of jump.
var j = 20;

// Object properties (position and velocity).
var px, py = 0, 0;
var vx, vy = 0, 0;

// This function is called every frame.
function update() {
    vy += g;

    if(/* Left is pressed. */) {
        vx -= m;
    }

    if(/* Right is pressed. */) {
        vx += m;
    }

    if(/* Shift is pressed. */) {
        // Make sure the object is on the floor.
        if(py == 500) {
            vy -= j;  // Jump.
        }
    }

    // Update position.
    px += vx;
    py += vy;

    // Check for collisions.
    px = Math.max(px, 0);
    px = Math.min(px, 500);
    py = Math.max(py, 0);
    py = Math.min(py, 500);

    // Bleed movement (causes object to slow down);
    vx *= 0.95;

    // Draw the object at position (px, py).
    draw(object, px, py);
}
This probably isn't exactly what you wanted and the algorithm is crap, but hopefully it will get you started.
