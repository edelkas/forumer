onClipEvent (load) {
	Fac = 0.5;
}
onClipEvent (enterFrame) {
	disx = _root.player._x-this._x;
	disy = _root.player._y-this._y;
	dis = Math.sqrt(disy*disy+disx*disx);
	SumRadius = 80*_xscale/100;
	direction = -Math.atan(disx/disy)*180/Math.PI;
	if (disy>0) {
  direction = -180+direction;
	}
	directionX = -SumRadius*Math.sin(direction*Math.PI/180)/2+disx;
	directionY = SumRadius*Math.cos(direction*Math.PI/180)/2+disy;
	_root.player.Speed = Math.sqrt(_root.player.xspeed*_root.player.xspeed+_root.player.yspeed*_root.player.yspeed);
	//----------Collision--------------------
	if (SumRadius>dis) {
  _root.player._x -= 2*directionX-disx;
  _root.player._y -= 2*directionY-disy;
  //---
  _root.player.xspeed = -_root.player.Speed*Math.sin(direction*Math.PI/180)*Fac;
  _root.player.yspeed = _root.player.Speed*Math.cos(direction*Math.PI/180)*Fac;
	}
	
}
