Public Function Factorial(ByVal lngFact As Long) As Long
    Dim i As Long
    Dim lngAnswer As Long: lngAnswer = 1
        For i = lngFact To 2 Step -1
            lngAnswer = lngAnswer * i
        Next
        Factorial = lngAnswer
End Function
